## 30.1 os 与 sys

```python
import os, sys

os.getcwd()            # 当前目录
os.environ["PATH"]    # 环境变量
os.path.join("a", "b")
os.path.exists("/tmp")

sys.argv               # 命令行参数
sys.path               # 模块搜索路径
sys.platform           # 'linux' / 'win' / 'darwin'
```

## 30.2 pathlib（推荐）

```python
from pathlib import Path

p = Path("/tmp/data.csv")
p.exists()
p.read_text()
p.write_text("hello")
p.parent
p.name
p.suffix      # ".csv"
p.with_suffix(".json")

# 遍历
for f in Path("data").glob("*.txt"):
    print(f)

# 目录
Path("logs").mkdir(parents=True, exist_ok=True)
```

## 30.3 json

```python
import json

data = {"name": "Alice", "age": 30}
s = json.dumps(data)          # 序列化为字符串
obj = json.loads(s)           # 解析

with open("data.json") as f:
    obj = json.load(f)

with open("data.json", "w") as f:
    json.dump(obj, f)
```

## 30.5 datetime

```python
from datetime import datetime, timedelta

now = datetime.now()
now.isoformat()
now + timedelta(days=7)
datetime.fromisoformat("2026-01-01T00:00:00")

# 时间差
delta = datetime(2026, 1, 10) - datetime(2026, 1, 1)
delta.days  # 9
```

## 30.6 collections

```python
from collections import Counter, defaultdict, deque

Counter("hello")         # Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})

d = defaultdict(list)
d["key"].append(1)       # 自动创建空列表

q = deque([1, 2, 3])
q.appendleft(0)
q.popleft()
```

## 30.7 itertools

```python
from itertools import chain, product, combinations, groupby

list(chain([1], [2], [3]))
list(product([0, 1], repeat=2))  # 笛卡尔积
list(combinations([1, 2, 3], 2))
```

## 30.8 functools

```python
from functools import lru_cache, reduce, partial

@lru_cache(maxsize=128)
def fib(n): ...

reduce(lambda a, b: a + b, [1, 2, 3, 4])  # 10

double = partial(lambda x, n=2: x * n, 5)
```
